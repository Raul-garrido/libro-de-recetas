// SINCRONIZACIÓN BLINDADA CONTRA BLOQUEOS async function syncWithDatabase() { if (!confirm("🔄 ¿Deseas sincronizar y actualizar las recetas desde GitHub? La página se recargará automáticamente.")) { return; } const syncBtn = document.getElementById('syncGithubBtn'); if (syncBtn) { syncBtn.textContent = "⏳ Sincronizando y actualizando..."; syncBtn.disabled = true; } // Creamos un controlador de tiempo de límite (timeout de 5 segundos) por si GitHub Pages tarda en responder const controller = new AbortController(); const timeoutId = setTimeout(() => controller.abort(), 5000); try { const response = await fetch('./recetas.json?v=' + Date.now(), { cache: 'no-store', headers: { 'Cache-Control': 'no-cache' }, signal: controller.signal }); clearTimeout(timeoutId); if (response.ok) { const jsonData = await response.json(); if (Array.isArray(jsonData)) { const req = indexedDB.deleteDatabase(dbName); req.onsuccess = async () => { await initDB(jsonData); window.location.reload(true); }; req.onerror = () => { window.location.reload(true); }; return; } } } catch (err) { // Si falla el fetch o hay timeout, forzamos igualmente la limpieza y recarga para aplicar los cambios locales/remotos } // Plan de respaldo si la red falla: borramos base de datos local y recargamos de todas formas const reqFallback = indexedDB.deleteDatabase(dbName); reqFallback.onsuccess = () => window.location.reload(true); reqFallback.onerror = () => window.location.reload(true); } // ARRANQUE AUTOMÁTICO DIRECTO DESDE recetas.json async function autoBootstrap() { initColorSystem(); if ('serviceWorker' in navigator) { navigator.serviceWorker.getRegistrations().then(registrations => { for(let registration of registrations) { registration.unregister(); } }); } try { const res = await fetch('./recetas.json?v=' + Date.now(), { cache: 'no-store' }); if (res.ok) { const data = await res.json(); if (Array.isArray(data) && data.length > 0) { await initDB(data); await loadRecipes(); renderGrid(); return; } } } catch(e) {} await initDB(); await loadRecipes(); renderGrid(); } autoBootstrap();
